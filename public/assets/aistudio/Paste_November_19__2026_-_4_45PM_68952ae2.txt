    (function initListingSync() {
      const inputTitle = document.getElementById('inputTitle');
      const inputPrice = document.getElementById('inputPrice');
      const inputLocation = document.getElementById('inputLocation');
      const inputCategory = document.getElementById('inputCategory');
      const inputEmail = document.getElementById('inputEmail');
      const inputDescription = document.getElementById('inputDescription');
      const charCount = document.getElementById('charCount');

      const previewTitle = document.getElementById('previewTitle');
      const previewPrice = document.getElementById('previewPrice');
      const previewLocation = document.getElementById('previewLocation');
      const previewCategory = document.getElementById('previewCategory');
      const previewEmail = document.getElementById('previewEmail');
      const previewDescription = document.getElementById('previewDescription');
      const previewConditionTag = document.getElementById('previewConditionTag');
      
      const photoPreviewContainer = document.getElementById('photoPreviewContainer');
      const previewImage = document.getElementById('previewImage');
      const previewEmptyPlaceholder = document.getElementById('previewEmptyPlaceholder');
      const removePhotoBtn = document.getElementById('removePhotoBtn');
      const imageUploadInput = document.getElementById('imageUploadInput');
      const photoCountLabel = document.getElementById('photoCountLabel');

      // Title synchronization
      inputTitle?.addEventListener('input', (e) => {
        const val = e.target.value.trim();
        previewTitle.textContent = val ? val : 'Title';
      });

      // Price synchronization with dynamic currency formatting
      inputPrice?.addEventListener('input', (e) => {
        let val = e.target.value.replace(/[^0-9.]/g, '');
        if (val) {
          previewPrice.textContent = '$' + val;
        } else {
          previewPrice.textContent = '$0';
        }
      });

      // Location synchronization
      inputLocation?.addEventListener('input', (e) => {
        const val = e.target.value.trim();
        previewLocation.textContent = val ? `in ${val}` : 'Location';
      });

      // Category synchronization
      inputCategory?.addEventListener('change', (e) => {
        previewCategory.textContent = e.target.value;
      });

      // Email synchronization
      inputEmail?.addEventListener('input', (e) => {
        const val = e.target.value.trim();
        previewEmail.textContent = val ? val : 'seller@example.com';
      });

      // Description synchronization & character counter
      inputDescription?.addEventListener('input', (e) => {
        const val = e.target.value;
        previewDescription.textContent = val.trim() ? val : 'No description provided.';
        if (charCount) {
          charCount.textContent = `${val.length} characters`;
        }
      });

      // Condition buttons toggle
      const conditionButtons = document.querySelectorAll('#conditionPills button');
      conditionButtons.forEach(btn => {
        btn.addEventListener('click', () => {
          conditionButtons.forEach(b => {
            b.classList.remove('bg-primary-fixed', 'text-primary', 'font-semibold');
            b.classList.add('bg-surface-container', 'text-on-surface-variant');
          });
          btn.classList.add('bg-primary-fixed', 'text-primary', 'font-semibold');
          btn.classList.remove('bg-surface-container', 'text-on-surface-variant');
          if (previewConditionTag) {
            previewConditionTag.textContent = btn.getAttribute('data-condition');
          }
        });
      });

      // Remove photo handler
      removePhotoBtn?.addEventListener('click', () => {
        photoPreviewContainer.classList.add('hidden');
        previewImage.classList.add('hidden');
        previewEmptyPlaceholder.classList.remove('hidden');
        photoCountLabel.textContent = '0 / 10 photos';
      });

      // Add image placeholder / simulated re-upload
      imageUploadInput?.addEventListener('change', (e) => {
        const file = e.target.files?.[0];
        if (file) {
          const reader = new FileReader();
          reader.onload = (event) => {
            const resultUrl = event.target?.result;
            const mainImg = document.getElementById('mainListingImage');
            if (mainImg) mainImg.src = resultUrl;
            previewImage.src = resultUrl;
            photoPreviewContainer.classList.remove('hidden');
            previewImage.classList.remove('hidden');
            previewEmptyPlaceholder.classList.add('hidden');
            photoCountLabel.textContent = '1 / 10 photos';
          };
          reader.readAsDataURL(file);
        }
      });

      // Desktop vs Mobile preview switcher
      const btnDesk = document.getElementById('previewDeviceDesktop');
      const btnMob = document.getElementById('previewDeviceMobile');
      const previewCardWrapper = document.getElementById('previewCardWrapper');

      btnDesk?.addEventListener('click', () => {
        btnDesk.classList.add('bg-surface-container', 'text-primary');
        btnDesk.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
        btnMob.classList.remove('bg-surface-container', 'text-primary');
        btnMob.classList.add('bg-surface-container-low', 'text-on-surface-variant');
        previewCardWrapper.classList.remove('max-w-sm', 'mx-auto');
      });

      btnMob?.addEventListener('click', () => {
        btnMob.classList.add('bg-surface-container', 'text-primary');
        btnMob.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
        btnDesk.classList.remove('bg-surface-container', 'text-primary');
        btnDesk.classList.add('bg-surface-container-low', 'text-on-surface-variant');
        previewCardWrapper.classList.add('max-w-sm', 'mx-auto');
      });
    })();
